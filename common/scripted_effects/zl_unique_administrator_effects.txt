# ============================================================================
#  大肥鱼 天枢执政 —— 形态切换脚本化效果
#  Mod: zl_unique_administrator
#  Stellaris Cetus v4.3.7
#
#      zl_switch_form_administrator   切为行政官
#      zl_switch_form_commander       切为指挥官
#      zl_switch_form_scientist       切为科学家
#
#  领袖本体的生成在 events 里完成（见 zl_leader.99）。
#
#  ── ★★ 配装设计 ★★ ──────────────────────────────────────────────────
#  本 mod 只保留 3 个自定义特质（三个形态各一个，不朽 + 自定义图标）。
#  其余【全部使用原版特质】。
#
#  每个形态 7 个，全部落在原版槽位上限内：
#      basic    2  +  veteran 3  +  destiny 1  +  subclass 1  =  7
#
#  ⚠ 为什么必须守上限（实测）：
#     配装超限（例如 veteran=4、destiny=3）时，把领袖【任命到内阁】
#     会触发槽位清理 —— 超出的特质被移除，并重新弹出"请选择特质"。
#
#  ⚠ 与"10 级要消耗 10 次选择"的冲突：
#     原版 LEADER_TRAIT_SELECTION_LEVELS = { 1..10 }，
#     但上限只允许 7 个特质。
#     解法：只给 7 个特质，等级用 add_skill_without_trait_selection
#     从 1 级直接补到 10 级（见 events 里的 create_leader）。
#
#  ── 选用的原版特质（三职业都可用 / 无寿命 / 无经验）──────────────────
#      leader_trait_gifted_2                          天赋异禀 II
#      leader_trait_intellectual                      知识分子
#      leader_trait_adventurous_spirit                冒险精神
#      leader_trait_legendary_scientific_diplomacy_3  科学外交 III
#      leader_trait_legendary_military_knowledge_3    军事知识 III
# ============================================================================


# ---------------------------------------------------------------------------
#  公共段：卸掉上一形态的全部特质 + 清标志位
#  在领袖作用域内调用。
# ---------------------------------------------------------------------------
zl_clear_forms_effect = {
	# ── 本 mod 的三个形态特质（形态 destiny）──────────────────────────
	remove_trait = zl_leader_trait_administrator
	remove_trait = zl_leader_trait_commander
	remove_trait = zl_leader_trait_scientist

	# ── 原版配装（basic ×2）──────────────────────────────────────────
	remove_trait = leader_trait_gifted_2
	remove_trait = leader_trait_intellectual
	# ── 原版配装（veteran ×3）────────────────────────────────────────
	remove_trait = leader_trait_adventurous_spirit
	remove_trait = leader_trait_legendary_scientific_diplomacy_3
	remove_trait = leader_trait_legendary_military_knowledge_3

	# ── 三个子职业 ──────────────────────────────────────────────────
	remove_trait = subclass_official_governor
	remove_trait = subclass_commander_admiral
	remove_trait = subclass_scientist_scholar

	# 三个形态标志位
	remove_leader_flag = zl_form_administrator
	remove_leader_flag = zl_form_commander
	remove_leader_flag = zl_form_scientist
}


# ---------------------------------------------------------------------------
#  行政官形态 —— 唯一带治理定位的形态
#  7 个：subclass 1 + basic 2 + veteran 3 + destiny 1
# ---------------------------------------------------------------------------
zl_switch_form_administrator = {
	event_target:zl_deepseek = {
		zl_clear_forms_effect = yes
		change_leader_class = official

		# 子职业（第 4 级的选择）
		add_trait = { trait = subclass_official_governor                     consume_selection = yes }
		# basic ×2（原版）
		add_trait = { trait = leader_trait_gifted_2                          consume_selection = yes }
		add_trait = { trait = leader_trait_intellectual                      consume_selection = yes }
		# veteran ×3（原版）
		add_trait = { trait = leader_trait_adventurous_spirit                consume_selection = yes }
		add_trait = { trait = leader_trait_legendary_scientific_diplomacy_3  consume_selection = yes }
		add_trait = { trait = leader_trait_legendary_military_knowledge_3    consume_selection = yes }
		# destiny ×1（本 mod 形态特质，不朽）
		add_trait = { trait = zl_leader_trait_administrator                  consume_selection = yes }

		set_leader_flag = zl_form_administrator
	}
}


# ---------------------------------------------------------------------------
#  指挥官形态 —— 舰队作战（无治理类加成）
# ---------------------------------------------------------------------------
zl_switch_form_commander = {
	event_target:zl_deepseek = {
		zl_clear_forms_effect = yes
		change_leader_class = commander

		add_trait = { trait = subclass_commander_admiral                     consume_selection = yes }
		add_trait = { trait = leader_trait_gifted_2                          consume_selection = yes }
		add_trait = { trait = leader_trait_intellectual                      consume_selection = yes }
		add_trait = { trait = leader_trait_adventurous_spirit                consume_selection = yes }
		add_trait = { trait = leader_trait_legendary_scientific_diplomacy_3  consume_selection = yes }
		add_trait = { trait = leader_trait_legendary_military_knowledge_3    consume_selection = yes }
		add_trait = { trait = zl_leader_trait_commander                      consume_selection = yes }

		set_leader_flag = zl_form_commander
	}
}

		add_trait = { trait = leader_trait_intellectual                      consume_selection = yes }
		add_trait = { trait = leader_trait_adventurous_spirit                consume_selection = yes }
		add_trait = { trait = leader_trait_legendary_scientific_diplomacy_3  consume_selection = yes }
		add_trait = { trait = leader_trait_legendary_military_knowledge_3    consume_selection = yes }
		add_trait = { trait = zl_leader_trait_scientist                      consume_selection = yes }

		set_leader_flag = zl_form_scientist
	}
