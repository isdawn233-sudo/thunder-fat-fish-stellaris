# ============================================================================
#  大肥鱼 天枢执政 —— 形态切换脚本化效果
#  Mod: zl_unique_administrator
#  Stellaris Cetus v4.3.7
#
#      zl_switch_form_administrator   切为行政官
#      zl_switch_form_commander       切为指挥官
#      zl_switch_form_scientist       切为科学家
#
#  领袖本体的生成在 events 里完成（见 zl_leader.99）。
#
#  ── ★★ 为什么配装只有 7 个特质 ★★ ────────────────────────────────────
#  原版 caps（common/defines/00_defines.txt）：
#      MAX_BASIC_TRAITS   = 2
#      MAX_VETERAN_TRAITS = 3
#  destiny 1 个、subclass 1 个。
#
#  ⚠ 实测踩坑：配装超限（veteran=4 / destiny=3）时，
#    把领袖【任命到内阁】会触发槽位清理 —— 超出的特质被移除，
#    并重新弹出"请选择特质"。所以超限绝对不能接受。
#
#  这与"10 级需要 10 次特质选择"冲突，解决方式：
#      只给 7 个特质（守住上限），
#      再用 add_skill_without_trait_selection 把等级补到 10。
#
#  配装构成（每个形态）：
#      通用 basic 2 + 通用 veteran 3 + 形态 veteran 1 + 形态 destiny 1
#      + 子职业（原版）1  =  8 个特质
#
#  ── 设计原则 ──────────────────────────────────────────────────────────
#  1) 不朽不靠寿命：形态特质带 immortal_leaders = yes（全局生效）
#  2) 不要经验加成：不含 species_leader_exp_gain
#  3) 只有行政官有治理定位：
#         行政官  modifier + planet_modifier + sector_modifier
#         指挥官  modifier + fleet_modifier
#         科学家  modifier
# ============================================================================


# ---------------------------------------------------------------------------
#  公共段：卸掉上一形态的全部特质 + 清标志位
# ---------------------------------------------------------------------------
zl_clear_forms_effect = {
	# ── 形态 destiny ×3 ──────────────────────────────────────────────
	remove_trait = zl_leader_trait_administrator
	remove_trait = zl_leader_trait_commander
	remove_trait = zl_leader_trait_scientist
	# ── 形态 veteran ×3 ─────────────────────────────────────────────
	remove_trait = zl_leader_trait_administrator_2
	remove_trait = zl_leader_trait_commander_2
	remove_trait = zl_leader_trait_scientist_2
	# ── 通用 basic ×2 ───────────────────────────────────────────────
	remove_trait = zl_leader_trait_gifted
	remove_trait = zl_leader_trait_resilient
	# ── 通用 veteran ×3 ─────────────────────────────────────────────
	remove_trait = zl_leader_trait_veteran
	remove_trait = zl_leader_trait_veteran_2
	# ── 三个子职业 ──────────────────────────────────────────────────
	remove_trait = subclass_official_governor
	remove_trait = subclass_commander_admiral
	remove_trait = subclass_scientist_scholar

	# 三个形态标志位
	remove_leader_flag = zl_form_administrator
	remove_leader_flag = zl_form_commander
	remove_leader_flag = zl_form_scientist
}


# ---------------------------------------------------------------------------
#  行政官形态 —— 唯一带治理定位的形态
#  8 个特质：subclass 1 + basic 2 + veteran 3 + destiny 1 + 形态 veteran 1
# ---------------------------------------------------------------------------
zl_switch_form_administrator = {
	event_target:zl_deepseek = {
		zl_clear_forms_effect = yes
		change_leader_class = official

		# 子职业（第 4 级的选择）
		add_trait = { trait = subclass_official_governor  consume_selection = yes }
		# basic ×2
		add_trait = { trait = zl_leader_trait_gifted      consume_selection = yes }
		add_trait = { trait = zl_leader_trait_resilient   consume_selection = yes }
		# veteran ×3（通用）
		add_trait = { trait = zl_leader_trait_veteran     consume_selection = yes }
		add_trait = { trait = zl_leader_trait_veteran_2   consume_selection = yes }
		# 形态 veteran（本职加成）
		add_trait = { trait = zl_leader_trait_administrator_2 consume_selection = yes }
		# destiny ×1（第 8 级的选择）
		add_trait = { trait = zl_leader_trait_administrator   consume_selection = yes }

		set_leader_flag = zl_form_administrator
	}
}


# ---------------------------------------------------------------------------
#  指挥官形态 —— 舰队作战（无治理类加成）
# ---------------------------------------------------------------------------
zl_switch_form_commander = {
	event_target:zl_deepseek = {
		zl_clear_forms_effect = yes
		change_leader_class = commander

		add_trait = { trait = subclass_commander_admiral   consume_selection = yes }
		add_trait = { trait = zl_leader_trait_gifted       consume_selection = yes }
		add_trait = { trait = zl_leader_trait_resilient    consume_selection = yes }
		add_trait = { trait = zl_leader_trait_veteran      consume_selection = yes }
		add_trait = { trait = zl_leader_trait_veteran_2    consume_selection = yes }
		add_trait = { trait = zl_leader_trait_commander_2  consume_selection = yes }
		add_trait = { trait = zl_leader_trait_commander    consume_selection = yes }

		set_leader_flag = zl_form_commander
	}
}


# ---------------------------------------------------------------------------
#  科学家形态 —— 研究与探索（无治理类加成）
# ---------------------------------------------------------------------------
zl_switch_form_scientist = {
	event_target:zl_deepseek = {
		zl_clear_forms_effect = yes
		change_leader_class = scientist

		add_trait = { trait = subclass_scientist_scholar  consume_selection = yes }
		add_trait = { trait = zl_leader_trait_gifted      consume_selection = yes }
		add_trait = { trait = zl_leader_trait_resilient   consume_selection = yes }
		add_trait = { trait = zl_leader_trait_veteran     consume_selection = yes }
		add_trait = { trait = zl_leader_trait_veteran_2   consume_selection = yes }
		add_trait = { trait = zl_leader_trait_scientist_2 consume_selection = yes }
		add_trait = { trait = zl_leader_trait_scientist   consume_selection = yes }

		set_leader_flag = zl_form_scientist
	}
}
